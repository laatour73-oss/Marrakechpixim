async function sha256(value) { const hash = await crypto.subtle.digest( "SHA-256", new TextEncoder().encode(value) ); return [...new Uint8Array(hash)] .map(x => x.toString(16).padStart(2, "0")) .join(""); } async function deriveSessionSecret(paymentId, key, env) { const cryptoKey = await crypto.subtle.importKey( "raw", new TextEncoder().encode(env.SESSION_HMAC_SECRET), { name: "HMAC", hash: "SHA-256" }, false, ["sign"] ); const signature = await crypto.subtle.sign( "HMAC", cryptoKey, new TextEncoder().encode(paymentId + "|" + key) ); return [...new Uint8Array(signature)] .map(x => x.toString(16).padStart(2, "0")) .join(""); } async function paypalToken(env) { const credentials = btoa( env.PAYPAL_CLIENT_ID + ":" + env.PAYPAL_CLIENT_SECRET ); const response = await fetch( env.PAYPAL_BASE_URL + "/v1/oauth2/token", { method: "POST", headers: { "Authorization": "Basic " + credentials, "Content-Type": "application/x-www-form-urlencoded" }, body: "grant_type=client_credentials" } ); const data = await response.json(); if (!response.ok || !data.access_token) { throw new Error("PAYPAL_AUTH_FAILED"); } return data.access_token; } async function paypalRequest(env, path, options = {}) { const token = await paypalToken(env); const response = await fetch( env.PAYPAL_BASE_URL + path, { ...options, headers: { "Authorization": "Bearer " + token, "Content-Type": "application/json", ...(options.headers || {}) } } ); const data = await response.json(); if (!response.ok) { throw new Error("PAYPAL_API_ERROR"); } return data; } // 1. CREATE CHECKOUT async function createCheckout(request, env) { const body = await request.json(); const key = request.headers.get("Idempotency-Key"); if (!key || !/^[a-zA-Z0-9_-]{16,120}$/.test(key)) { return json({ error: "Invalid idempotency key" }, 400); } if (body.method !== "paypal") { return json({ error: "Payment provider is not configured" }, 503); } const existing = await env.DB.prepare( "SELECT * FROM browse_payments WHERE idempotency_key = ?" ).bind(key).first(); let paymentId, sessionId, secret; if (existing) { paymentId = existing.payment_id; sessionId = existing.session_id; secret = await deriveSessionSecret(paymentId, key, env); const session = await env.DB.prepare( "SELECT secret_hash FROM browse_sessions WHERE session_id = ?" ).bind(sessionId).first(); if (!session || await sha256(secret) !== session.secret_hash) { return json({ error: "Session verification failed" }, 403); } if (existing.paypal_order_id) { const order = await paypalRequest( env, "/v2/checkout/orders/" + encodeURIComponent(existing.paypal_order_id) ); const approval = order.links?.find(link => link.rel === "approve"); return json({ success: true, duplicate: true, orderId: existing.paypal_order_id, status: existing.status, checkoutUrl: approval?.href || null, sessionId, sessionSecret: secret }); } } else { paymentId = crypto.randomUUID(); sessionId = paymentId; secret = await deriveSessionSecret(paymentId, key, env); const secretHash = await sha256(secret); await env.DB.batch([ env.DB.prepare(` INSERT INTO browse_sessions (session_id, secret_hash, status, expires_at, created_at) VALUES (?, ?, 'PENDING', 0, ?) `).bind(sessionId, secretHash, now()), env.DB.prepare(` INSERT INTO browse_payments (payment_id, session_id, provider, idempotency_key, amount, currency, status, created_at) VALUES (?, ?, 'paypal', ?, '3.00', 'USD', 'PENDING', ?) `).bind(paymentId, sessionId, key, now()) ]); } try { const order = await paypalRequest( env, "/v2/checkout/orders", { method: "POST", headers: { "PayPal-Request-Id": paymentId }, body: JSON.stringify({ intent: "CAPTURE", purchase_units: [{ reference_id: paymentId, custom_id: paymentId, invoice_id: paymentId, amount: { currency_code: "USD", value: "3.00" } }], payment_source: { paypal: { experience_context: { return_url: env.PAYMENT_RETURN_URL, cancel_url: env.PAYMENT_CANCEL_URL, user_action: "PAY_NOW" } } } }) } ); const approval = order.links?.find(link => link.rel === "approve"); if (!order.id || !approval?.href) throw new Error("INVALID_PAYPAL_ORDER"); await env.DB.prepare(` UPDATE browse_payments SET paypal_order_id = ? WHERE payment_id = ? AND status = 'PENDING' `).bind(order.id, paymentId).run(); return json({ success: true, orderId: order.id, paymentId, sessionId, sessionSecret: secret, amount: "3.00", currency: "USD", minutes: 20, status: "PENDING", checkoutUrl: approval.href }, 201); } catch (error) { console.error("Checkout error:", error.message); return json({ error: "Unable to create checkout" }, 502); } } // 2. CAPTURE PAYMENT async function capturePayment(request, env) { const body = await request.json(); const secret = request.headers.get("X-Session-Secret"); if (!secret || !body.orderId) return json({ error: "Missing authorization" }, 401); const payment = await env.DB.prepare(`SELECT * FROM browse_payments WHERE paypal_order_id = ?`).bind(body.orderId).first(); if (!payment) return json({ error: "Payment not found" }, 404); const session = await env.DB.prepare(`SELECT * FROM browse_sessions WHERE session_id = ?`).bind(payment.session_id).first(); if (!session || await sha256(secret) !== session.secret_hash) return json({ error: "Unauthorized session" }, 403); if (payment.status === "COMPLETED") return json({ success: true, status: "COMPLETED" }); const result = await paypalRequest( env, "/v2/checkout/orders/" + encodeURIComponent(payment.paypal_order_id) + "/capture", { method: "POST", headers: { "PayPal-Request-Id": payment.payment_id + "-capture" }, body: "{}" } ); return json({ success: true, captureStatus: result.status, message: "Waiting for verified webhook confirmation" }); } // 3. VERIFY PAYPAL WEBHOOK async function verifyPayPalWebhook(request, event, env) { const headers = request.headers; const payload = { auth_algo: headers.get("paypal-auth-algo"), cert_url: headers.get("paypal-cert-url"), transmission_id: headers.get("paypal-transmission-id"), transmission_sig: headers.get("paypal-transmission-sig"), transmission_time: headers.get("paypal-transmission-time"), webhook_id: env.PAYPAL_WEBHOOK_ID, webhook_event: event }; if (!payload.auth_algo || !payload.cert_url || !payload.transmission_id || !payload.transmission_sig || !payload.transmission_time || !payload.webhook_id) { return false; } const result = await paypalRequest(env, "/v1/notifications/verify-webhook-signature", { method: "POST", body: JSON.stringify(payload) }); return result.verification_status === "SUCCESS"; } // 4. APPLY PAYMENT async function applyPayment(event, env) { if (event.event_type !== "PAYMENT.CAPTURE.COMPLETED") return json({ received: true, ignored: true }); const capture = event.resource; const orderId = capture.supplementary_data?.related_ids?.order_id; if (!orderId || capture.status !== "COMPLETED") return json({ error: "Invalid capture" }, 400); const order = await paypalRequest(env, "/v2/checkout/orders/" + encodeURIComponent(orderId)); if (order.status !== "COMPLETED") return json({ error: "Order not completed" }, 409); const unit = order.purchase_units?.[0]; const paymentId = unit?.custom_id; const amount = unit?.payments?.captures?.[0]?.amount; if (!paymentId || amount?.value !== "3.00" || amount?.currency_code !== "USD") { return json({ error: "Amount verification failed" }, 400); } const payment = await env.DB.prepare(`SELECT * FROM browse_payments WHERE payment_id = ? AND paypal_order_id = ?`).bind(paymentId, orderId).first(); if (!payment) return json({ error: "Unknown payment" }, 404); await env.DB.batch([ env.DB.prepare(` UPDATE browse_sessions SET status = 'ACTIVE', expires_at = CASE WHEN expires_at > ? THEN expires_at + 1200 ELSE ? + 1200 END WHERE session_id = ? AND EXISTS (SELECT 1 FROM browse_payments WHERE payment_id = ? AND grant_applied = 0 AND status = 'PENDING') `).bind(now(), now(), payment.session_id, paymentId), env.DB.prepare(`UPDATE browse_payments SET status = 'COMPLETED', grant_applied = 1 WHERE payment_id = ? AND status = 'PENDING' AND grant_applied = 0`).bind(paymentId), env.DB.prepare(`INSERT INTO webhook_events (event_id, provider, received_at) VALUES (?, 'paypal', ?) ON CONFLICT(event_id) DO NOTHING`).bind(event.id, now()) ]); return json({ received: true, payment: "processed" }); } // 5. WEBHOOK ROUTER async function webhookRouter(request, env, provider) { if (provider !== "paypal") return json({ error: "Provider not configured" }, 501); const raw = await request.text(); if (raw.length > 100000) return json({ error: "Payload too large" }, 413); let event; try { event = JSON.parse(raw); } catch { return json({ error: "Invalid JSON" }, 400); } try { const verified = await verifyPayPalWebhook(request, event, env); if (!verified) return json({ error: "Invalid signature" }, 401); if (!event.id) return json({ error: "Missing event ID" }, 400); const existing = await env.DB.prepare(`SELECT event_id FROM webhook_events WHERE event_id = ?`).bind(event.id).first(); if (existing) return json({ received: true, duplicate: true }); return await applyPayment(event, env); } catch (error) { console.error("Webhook error:", error.message); return json({ error: "Webhook processing failed" }, 500); } } // 6.
